file(GLOB_RECURSE UCM_POSIX_STORE_CC_SOURCE_FILES "./cc/*.cc")
add_library(posixstore SHARED ${UCM_POSIX_STORE_CC_SOURCE_FILES})
target_include_directories(posixstore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cc)

find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(LIBURING IMPORTED_TARGET liburing)
endif()
if(TARGET PkgConfig::LIBURING)
    set(UCM_LIBURING_TARGET PkgConfig::LIBURING)
endif()
find_path(LIBURING_INCLUDE_DIR NAMES liburing.h)
find_library(LIBURING_LIBRARY NAMES uring)
if(LIBURING_INCLUDE_DIR AND LIBURING_LIBRARY)
    if(NOT TARGET ucm_liburing)
        add_library(ucm_liburing INTERFACE IMPORTED)
        target_include_directories(ucm_liburing INTERFACE ${LIBURING_INCLUDE_DIR})
    endif()
    target_link_libraries(ucm_liburing INTERFACE ${LIBURING_LIBRARY})
    if(NOT UCM_LIBURING_TARGET)
        set(UCM_LIBURING_TARGET ucm_liburing)
    endif()
endif()
if(NOT UCM_LIBURING_TARGET)
    message(FATAL_ERROR
        "liburing not found; install liburing-dev (or liburing-devel) to build the posix io_uring engine")
endif()

target_link_libraries(posixstore PUBLIC storeintf infra_logger metrics ${UCM_LIBURING_TARGET})
if(LIBURING_LIBRARY)
    target_link_libraries(posixstore PUBLIC ${LIBURING_LIBRARY})
endif()
set_target_properties(posixstore PROPERTIES
    INSTALL_RPATH "$ORIGIN/../../shared/metrics"
)
if(BUILD_UNIT_TESTS)
    target_compile_definitions(posixstore PUBLIC UCM_ENABLE_TEST_HOOKS)
endif()

file(RELATIVE_PATH INSTALL_REL_PATH ${UCM_ROOT_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
install(TARGETS posixstore LIBRARY DESTINATION ${INSTALL_REL_PATH} COMPONENT ucm)
